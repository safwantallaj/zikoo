} // د) استرجاع قائمة الأصناف من المستودع if (action === "GET_INVENTORY_ITEMS") { ensureInventorySheetsExist(ss); var stockSheet = ss.getSheetByName("المخزن والأصناف"); var rows = stockSheet.getDataRange().getValues(); } } function updateStockQuantity(stockSheet, itemCode, itemName, qty, isSales, costPrice, salePrice) { var data = stockSheet.getDataRange().getValues(); var foundRow = -1; for (var i = 1; i < data.length; i++) { if (String(data[i][0]) === String(itemCode) || String(data[i][1]) === String(itemName)) { foundRow = i + 1; break; } } if (foundRow > 0) { var inQty = Number(stockSheet.getRange(foundRow, 8).getValue()) || 0; var outQty = Number(stockSheet.getRange(foundRow, 9).getValue()) || 0; var initQty = Number(stockSheet.getRange(foundRow, 7).getValue()) || 0; if (isSales) { outQty += qty; stockSheet.getRange(foundRow, 9).setValue(outQty); } else { inQty += qty; stockSheet.getRange(foundRow, 8).setValue(inQty); } var currentBal = initQty + inQty - outQty; stockSheet.getRange(foundRow, 10).setValue(currentBal); } else { // صنف جديد يسجل لأول مرة في المستودع var inQ = isSales ? 0 : qty; var outQ = isSales ? qty : 0; var cur = inQ - outQ; var newCode = itemCode || ("ITM-" + Math.floor(100 + Math.random() * 900)); stockSheet.appendRow([ newCode, itemName, 'حبة', '', costPrice, salePrice, 0, inQ, outQ, cur, 5 ]); } }
